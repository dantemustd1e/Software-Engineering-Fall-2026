# Expense calculator

A small command line tool to keep track of personal expenses in KZT.

## How to run

Python 3.10 or newer is required.

    python main.py add --date 2026-11-01 --category food --amount 1200 --note coffee
    python main.py list
    python main.py total
    python main.py by-category
    python main.py monthly --month 2026-11
    python main.py export --format text --out report.txt

The default data file is `expenses.csv` in the current folder.

Dates should be written as YYYY-MM-DD. Amounts are in KZT.

## Tests

    python -m unittest discover -v

"""In-memory expense book. Amounts are stored in KZT."""

from dataclasses import dataclass


@dataclass
class Expense:
    date: str
    category: str
    amount: float
    note: str = ""


class ExpenseBook:
    def __init__(self, items=None):
        if items is None:
            self.items = []
        else:
            self.items = list(items)

    def add(self, date, category, amount, note=""):
        expense = Expense(str(date), str(category), float(amount), str(note))
        self.items.append(expense)
        return expense

    def list_all(self):
        return list(self.items)

    def total(self):
        value = 0.0
        for expense in self.items:
            value = value + expense.amount
        return value

    def totals_by_category(self):
        totals = {}
        for expense in self.items:
            if expense.category in totals:
                totals[expense.category] = totals[expense.category] + expense.amount
            else:
                totals[expense.category] = expense.amount
        return totals

    def monthly_summary(self, year_month):
        selected = []
        for expense in self.items:
            if expense.date.startswith(year_month):
                selected.append(expense)
        month_book = ExpenseBook(selected)
        return {
            "month": year_month,
            "count": len(selected),
            "total": month_book.total(),
            "by_category": month_book.totals_by_category(),
            "items": selected,
        }

    def split_evenly(self, n):
        return self.total() / n

"""Load and save expenses as a simple CSV file."""

import csv

from expenses import ExpenseBook


def load_csv(path):
    book = ExpenseBook()
    f = open(path, "r")
    reader = csv.reader(f)
    first = True
    for row in reader:
        if first:
            first = False
            if row and row[0].lower() == "date":
                continue
        date, category, amount, note = row
        book.add(date, category, amount, note)
    f.close()
    return book


def save_csv(path, book):
    f = open(path, "w")
    writer = csv.writer(f)
    writer.writerow(["date", "category", "amount", "note"])
    for expense in book.list_all():
        writer.writerow([expense.date, expense.category, expense.amount, expense.note])
    f.close()

"""Text and CSV reports for an expense book."""

import csv


def format_line(expense):
    return f"{expense.date}  {expense.category:12}  {expense.amount:10.2f}  {expense.note}"


def export_text(book, path=None):
    lines = ["Expense report", "-" * 40]
    for expense in book.list_all():
        lines.append(format_line(expense))
    lines.append("-" * 40)
    lines.append("Total: {:.2f}".format(book.total()))
    lines.append("")
    lines.append("By category:")
    totals = book.totals_by_category()
    for category in totals:
        lines.append(f"  {category}: {totals[category]:.2f}")
    text = "\n".join(lines) + "\n"
    if path:
        f = open(path, "w")
        f.write(text)
        f.close()
    return text


def export_csv(book, path):
    f = open(path, "w")
    writer = csv.writer(f)
    writer.writerow(["date", "category", "amount", "note"])
    for expense in book.list_all():
        writer.writerow([expense.date, expense.category, expense.amount, expense.note])
    f.close()


def export_monthly(book, month, path=None):
    summary = book.monthly_summary(month)
    lines = [f"Monthly report for {month}", "-" * 40]
    for expense in summary["items"]:
        lines.append(format_line(expense))
    lines.append("-" * 40)
    lines.append("Total: {:.2f}".format(summary["total"]))
    lines.append("Count: {}".format(summary["count"]))
    lines.append("By category:")
    for category in summary["by_category"]:
        lines.append(f"  {category}: {summary['by_category'][category]:.2f}")
    text = "\n".join(lines) + "\n"
    if path:
        f = open(path, "w")
        f.write(text)
        f.close()
    return text


def export(book, fmt, path=None, month=None):
    if fmt == "text" or fmt == "txt":
        return export_text(book, path)
    elif fmt == "csv":
        if not path:
            raise ValueError("csv export needs a path")
        export_csv(book, path)
        return path
    elif fmt == "monthly":
        if month is None:
            raise ValueError("monthly export needs a month")
        return export_monthly(book, month, path)
    else:
        raise ValueError("unknown format: " + str(fmt))

"""Command line entry point for the expense calculator."""

import argparse
import os
import sys

from expenses import ExpenseBook
from storage import load_csv, save_csv
import report

DEFAULT_FILE = "expenses.csv"


def load_or_empty(path):
    if os.path.exists(path):
        return load_csv(path)
    return ExpenseBook()


def cmd_add(args):
    book = load_or_empty(args.file)
    book.add(args.date, args.category, args.amount, args.note)
    save_csv(args.file, book)
    print("added:", args.date, args.category, args.amount, args.note)


def cmd_list(args):
    book = load_csv(args.file)
    for expense in book.list_all():
        print(expense.date, expense.category, expense.amount, expense.note)
    print("total:", book.total())


def cmd_total(args):
    book = load_csv(args.file)
    print(book.total())


def cmd_by_category(args):
    book = load_csv(args.file)
    totals = book.totals_by_category()
    for category in totals:
        print(f"{category}: {totals[category]}")


def cmd_monthly(args):
    book = load_csv(args.file)
    summary = book.monthly_summary(args.month)
    print("month:", summary["month"])
    print("count:", summary["count"])
    print("total:", summary["total"])
    for category in summary["by_category"]:
        print(f"  {category}: {summary['by_category'][category]}")


def cmd_export(args):
    book = load_csv(args.file)
    out = report.export(book, args.format, path=args.out, month=args.month)
    if args.format == "text" or args.format == "txt" or args.format == "monthly":
        if args.out:
            print("wrote", args.out)
        else:
            print(out, end="")
    else:
        print("wrote", args.out)


def cmd_split(args):
    book = load_csv(args.file)
    print("each person pays:", book.split_evenly(args.people))


def build_parser():
    parser = argparse.ArgumentParser(description="Simple expense calculator")
    parser.add_argument("--file", default=DEFAULT_FILE, help="csv file with expenses")
    sub = parser.add_subparsers(dest="command")

    add_p = sub.add_parser("add")
    add_p.add_argument("--date", required=True)
    add_p.add_argument("--category", required=True)
    add_p.add_argument("--amount", required=True)
    add_p.add_argument("--note", default="")
    add_p.set_defaults(func=cmd_add)

    sub.add_parser("list").set_defaults(func=cmd_list)
    sub.add_parser("total").set_defaults(func=cmd_total)
    sub.add_parser("by-category").set_defaults(func=cmd_by_category)

    mon_p = sub.add_parser("monthly")
    mon_p.add_argument("--month", required=True, help="YYYY-MM")
    mon_p.set_defaults(func=cmd_monthly)

    exp_p = sub.add_parser("export")
    exp_p.add_argument("--format", default="text")
    exp_p.add_argument("--out", default=None)
    exp_p.add_argument("--month", default=None)
    exp_p.set_defaults(func=cmd_export)

    split_p = sub.add_parser("split")
    split_p.add_argument("--people", type=int, required=True)
    split_p.set_defaults(func=cmd_split)
    return parser


def main(argv=None):
    parser = build_parser()
    args = parser.parse_args(argv)
    if not getattr(args, "command", None):
        parser.print_help()
        return 1
    args.func(args)
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""Basic tests for the expense calculator."""

import os
import tempfile
import unittest

from expenses import ExpenseBook
from storage import load_csv, save_csv
import report


class ExpenseBookTests(unittest.TestCase):
    def setUp(self):
        self.book = ExpenseBook()
        self.book.add("2026-11-01", "food", 1200, "coffee")
        self.book.add("2026-11-01", "food", 3500, "lunch at canteen")
        self.book.add("2026-11-03", "transport", 300, "bus")
        self.book.add("2026-10-20", "study", 4500, "printouts")

    def test_add_and_list(self):
        items = self.book.list_all()
        self.assertEqual(len(items), 4)
        self.assertEqual(items[0].date, "2026-11-01")
        self.assertEqual(items[0].category, "food")
        self.assertEqual(items[0].amount, 1200)
        self.assertEqual(items[0].note, "coffee")

    def test_total(self):
        self.assertEqual(self.book.total(), 1200 + 3500 + 300 + 4500)

    def test_totals_by_category(self):
        totals = self.book.totals_by_category()
        self.assertEqual(totals["food"], 4700)
        self.assertEqual(totals["transport"], 300)
        self.assertEqual(totals["study"], 4500)

    def test_monthly_summary(self):
        november = self.book.monthly_summary("2026-11")
        self.assertEqual(november["count"], 3)
        self.assertEqual(november["total"], 5000)
        self.assertEqual(november["by_category"]["food"], 4700)

    def test_empty_book(self):
        empty = ExpenseBook()
        self.assertEqual(empty.list_all(), [])
        self.assertEqual(empty.total(), 0)
        self.assertEqual(empty.totals_by_category(), {})

    def test_csv_roundtrip(self):
        fd, path = tempfile.mkstemp(suffix=".csv")
        os.close(fd)
        try:
            save_csv(path, self.book)
            loaded = load_csv(path)
            self.assertEqual(len(loaded.list_all()), 4)
            self.assertEqual(loaded.total(), self.book.total())
            self.assertEqual(loaded.list_all()[1].note, "lunch at canteen")
        finally:
            os.remove(path)

    def test_export_text_contains_total(self):
        text = report.export_text(self.book)
        self.assertIn("Expense report", text)
        self.assertIn("Total:", text)
        self.assertIn("coffee", text)

    def test_export_csv_writes_file(self):
        fd, path = tempfile.mkstemp(suffix=".csv")
        os.close(fd)
        try:
            report.export(self.book, "csv", path=path)
            with open(path) as handle:
                content = handle.read()
            self.assertIn("date,category,amount,note", content)
            self.assertIn("bus", content)
        finally:
            os.remove(path)


if __name__ == "__main__":
    unittest.main()
